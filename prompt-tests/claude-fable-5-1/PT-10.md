from mathutils import Vector
from mathutils.bvhtree import BVHTree

OUT = os.path.dirname(bpy.data.filepath) or os.getcwd()
SEG = 96          # segments around the axis
PITCH, DEPTH, FLANK = 1.27, 0.35, 0.35   # groove pitch, radial depth, 45 deg flank run (mm)
R_SHANK = 3.2

def groove_run(x0, x1, r):
    """Trapezoid grooves, 45 deg flanks, from x0 to x1. Returns (x, r) list."""
    pts = []
    x = x0
    while x + PITCH <= x1 + 1e-6:
        pts += [(x, r), (x + FLANK, r - DEPTH), (x + PITCH - FLANK, r - DEPTH), (x + PITCH, r)]
        x += PITCH
    return pts

def dome(h, rb, n=12):
    """Spherical cap, apex at x=0, base radius rb at x=h."""
    R = (rb*rb + h*h) / (2*h)
    pts = []
    for i in range(n + 1):
        t = i / n
        x = t * h
        r = math.sqrt(max(R*R - (R - x)**2, 0.0))
        pts.append((x, r))
    return pts

# ---------- profile (x along axis, r radius), all mm, measured from Fig. 3 ----------
P = []
P += dome(3.5, 5.65)                  # head 14, dia 11.3
P += [(4.0, 5.65)]                    # short head rim
P += [(4.0, 4.55), (5.6, 4.55)]       # washer 39/40, OD 9.1, 1.6 thick
P += [(5.6, R_SHANK)]                 # shank 15 starts
P += groove_run(10.9, 16.9, R_SHANK)  # lock grooves 12/13 up to collar
P += [(16.9, R_SHANK), (16.9, 5.05)]  # collar 41 square end, OD 10.1
P += [(21.4, 5.05), (22.4, 4.3)]      # chamfer 42
P += [(22.4, R_SHANK)]                # pintail 22
P += groove_run(25.4, 28.9, R_SHANK)  # pull grooves
P += [(28.9, R_SHANK), (29.4, 2.4), (29.7, 1.2), (29.7, 0.0)]  # rounded tail end
# dedupe consecutive identical points

for p in P[1:]:

        Q.append(p)

sc.unit_settings.length_unit = 'MILLIMETERS'

bmesh.ops.remove_doubles(bm, verts=bm.verts, dist=1e-4)
bmesh.ops.dissolve_degenerate(bm, dist=1e-5, edges=bm.edges)
bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
bm.to_mesh(mesh); bm.free()

bpy.context.view_layer.objects.active = obj
bpy.ops.object.transform_apply(location=True, rotation=True, scale=True)

# normals outside: signed volume must be positive

# thin walls: ray from each face centre inward along -normal, hit distance < 1.2 mm

# key dimension: shank diameter at x = 8.0 mm
